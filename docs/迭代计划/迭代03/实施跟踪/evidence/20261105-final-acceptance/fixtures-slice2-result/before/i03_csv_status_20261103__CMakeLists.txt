cmake_minimum_required(VERSION 3.16)
project(csv_status_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(NOT EXISTS "${BASELINE_SOURCE}")
  message(FATAL_ERROR "Set BASELINE_SOURCE to the saved pre-change WtDtHelper.cpp")
endif()
set(COMMON_INCLUDES "${REPO}" "${REPO}/src" "${REPO}/src/WtDtHelper" /opt/homebrew/Cellar/boost/1.88.0/include /opt/homebrew/Cellar/zstd/1.5.7_1/include /opt/homebrew/Cellar/spdlog/1.15.3/include /opt/homebrew/Cellar/fmt/11.2.0/include /opt/homebrew/Cellar/googletest/1.17.0/include)
set(COMMON_LIBS /tmp/wt-i03-r41-drain/libs/libWTSTools.a /tmp/wt-i03-r41-drain/libs/libWTSUtils.a /opt/homebrew/Cellar/boost/1.88.0/lib/libboost_filesystem.dylib /opt/homebrew/Cellar/boost/1.88.0/lib/libboost_thread.dylib /opt/homebrew/Cellar/zstd/1.5.7_1/lib/libzstd.1.5.7.dylib iconv dl pthread)
add_executable(csv_baseline "${BASELINE_SOURCE}" baseline.cpp)
add_executable(csv_status "${REPO}/src/WtDtHelper/WtDtHelper.cpp" check.cpp)
foreach(target csv_baseline csv_status)
  target_include_directories(${target} PRIVATE ${COMMON_INCLUDES})
  target_include_directories(${target} SYSTEM PRIVATE /usr/local/include)
  target_compile_definitions(${target} PRIVATE FMT_HEADER_ONLY SPDLOG_FMT_EXTERNAL WT_DT_HELPER_CLOSE_TEST_HOOKS)
  target_compile_options(${target} PRIVATE "SHELL:-include stdlib.h" "SHELL:-include ${REPO}/src/Share/fmt_enum_compat.h" -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${target} PRIVATE -fsanitize=address,undefined -fprofile-instr-generate)
  target_link_libraries(${target} PRIVATE ${COMMON_LIBS})
endforeach()
target_link_libraries(csv_status PRIVATE /opt/homebrew/Cellar/googletest/1.17.0/lib/libgtest_main.a /opt/homebrew/Cellar/googletest/1.17.0/lib/libgtest.a)
# No test hooks: verify the actual shared-library export surface separately.
add_library(csv_status_exports SHARED "${REPO}/src/WtDtHelper/WtDtHelper.cpp")
target_include_directories(csv_status_exports PRIVATE ${COMMON_INCLUDES})
target_include_directories(csv_status_exports SYSTEM PRIVATE /usr/local/include)
target_compile_definitions(csv_status_exports PRIVATE FMT_HEADER_ONLY SPDLOG_FMT_EXTERNAL)
target_compile_options(csv_status_exports PRIVATE "SHELL:-include stdlib.h" "SHELL:-include ${REPO}/src/Share/fmt_enum_compat.h")
target_link_options(csv_status_exports PRIVATE -fprofile-instr-generate)
target_link_libraries(csv_status_exports PRIVATE ${COMMON_LIBS})
set_target_properties(csv_status_exports PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
