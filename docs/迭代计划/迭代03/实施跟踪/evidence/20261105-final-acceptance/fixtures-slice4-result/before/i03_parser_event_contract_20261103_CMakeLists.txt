# name source-dir output-dir [hook-definition]
function(wt_module target name srcdir outdir)
 add_library(${target} SHARED "${srcdir}/${name}.cpp")
 target_include_directories(${target} PRIVATE "${ROOT}/src/${name}")
 set_target_properties(${target} PROPERTIES OUTPUT_NAME ${name} LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${outdir}")
 if(ARGC GREATER 4)
  target_compile_definitions(${target} PRIVATE ${ARGV4})
 endif()
 set(MODULES ${MODULES} ${target} PARENT_SCOPE)
endfunction()
# Current sources (green).
