cmake_minimum_required(VERSION 3.16)
project(i03_parser_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(WT_BUILD "" CACHE PATH "Build providing static dependencies")
foreach(mode RANGE 0 12)
  add_library(Pvt2${mode} SHARED provider.cpp)
  target_compile_definitions(Pvt2${mode} PRIVATE P_MODE=${mode})
  target_include_directories(Pvt2${mode} PRIVATE "${ROOT}/src" /opt/homebrew/include /opt/homebrew/Cellar/boost/1.88.0/include)
  set_target_properties(Pvt2${mode} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/parsers")
endforeach()
add_executable(i03_parser_v2 check.cpp "${ROOT}/src/WtCore/ParserAdapter.cpp")
target_include_directories(i03_parser_v2 PRIVATE "${ROOT}/src" /opt/homebrew/include /opt/homebrew/Cellar/boost/1.88.0/include)
target_compile_definitions(i03_parser_v2 PRIVATE FMT_HEADER_ONLY SPDLOG_FMT_EXTERNAL)
target_compile_options(i03_parser_v2 PRIVATE -include "${ROOT}/src/Share/fmt_enum_compat.h")
target_link_options(i03_parser_v2 PRIVATE -fprofile-instr-generate)
target_link_directories(i03_parser_v2 PRIVATE /opt/homebrew/lib)
target_link_libraries(i03_parser_v2 PRIVATE "${WT_BUILD}/build_x64/Debug/libs/libWtCore.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSTools.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSUtils.a" /opt/homebrew/lib/libgtest_main.a /opt/homebrew/lib/libgtest.a boost_filesystem boost_thread pthread dl /usr/local/lib/libiconv.dylib)
