cmake_minimum_required(VERSION 3.16)
project(i03_monitor_date LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(WT_BUILD "" CACHE PATH "Independent prebuilt WtDtCore/WTSTools/WTSUtils")
set(INCS "${ROOT}/src" /opt/homebrew/Cellar/boost/1.88.0/include /opt/homebrew/include /usr/local/include)
set(LIBS "${WT_BUILD}/build_x64/Debug/libs/libWtDtCore.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSTools.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSUtils.a" /opt/homebrew/lib/libgtest_main.a /opt/homebrew/lib/libgtest.a boost_filesystem boost_thread nanomsg pthread dl /usr/local/lib/libiconv.dylib)
add_library(MonitorSpy SHARED provider.cpp)
target_include_directories(MonitorSpy PRIVATE ${INCS})
add_executable(monitor_date check.cpp "${ROOT}/src/WtDtCore/StateMonitor.cpp" "${ROOT}/src/WtDtPorter/WtDtRunner.cpp" "${ROOT}/src/WtDtPorter/ExpDumper.cpp" "${ROOT}/src/WtDtPorter/ExpParser.cpp" "${ROOT}/src/WTSTools/WTSBaseDataMgr.cpp")
add_executable(q05_monitor_control "${ROOT}/src/TestUnits/coverage/q05_dtsrv/q05_state_monitor_test.cpp" "${ROOT}/src/WtDtCore/StateMonitor.cpp" "${ROOT}/src/WTSTools/WTSBaseDataMgr.cpp")
foreach(target monitor_date q05_monitor_control)
  target_include_directories(${target} PRIVATE ${INCS})
  target_compile_definitions(${target} PRIVATE FMT_HEADER_ONLY SPDLOG_FMT_EXTERNAL)
  target_compile_options(${target} PRIVATE "SHELL:-include cstdlib" "SHELL:-include ${ROOT}/src/Share/fmt_enum_compat.h" -fsanitize=undefined -fno-sanitize-recover=all)
  target_link_options(${target} PRIVATE -fsanitize=undefined -fprofile-instr-generate "LINKER:-rpath,/opt/homebrew/lib" "LINKER:-rpath,/usr/local/lib")
  target_link_directories(${target} PRIVATE /opt/homebrew/lib /usr/local/lib)
  target_link_libraries(${target} PRIVATE ${LIBS})
endforeach()
