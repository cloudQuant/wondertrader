cmake_minimum_required(VERSION 3.16)
project(i03_parser_shm_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(WT_BUILD "" CACHE PATH "Build providing static dependencies")
add_library(ParserShm SHARED "${ROOT}/src/ParserShm/ParserShm.cpp")
set_target_properties(ParserShm PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/parsers")
add_library(ParserShmFault SHARED "${ROOT}/src/ParserShm/ParserShm.cpp")
target_compile_definitions(ParserShmFault PRIVATE WT_PARSERSHM_TEST_HOOKS)
set_target_properties(ParserShmFault PROPERTIES OUTPUT_NAME ParserShm LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fault/parsers")
add_executable(i03_parser_shm_v2 check.cpp "${ROOT}/src/WtCore/ParserAdapter.cpp")
add_executable(i03_parser_shm_legacy "${ROOT}/src/ParserShm/ParserShm.cpp" "${ROOT}/src/TestUnits/coverage/t_ParserShm/t5_shm_parser_test.cpp" "${ROOT}/src/TestUnits/coverage/t_ParserShm/r76_shm_boundary_test.cpp")
foreach(target ParserShm ParserShmFault i03_parser_shm_v2 i03_parser_shm_legacy)
 target_include_directories(${target} PRIVATE "${ROOT}/src" /opt/homebrew/include /opt/homebrew/Cellar/boost/1.88.0/include)
 target_compile_definitions(${target} PRIVATE FMT_HEADER_ONLY SPDLOG_FMT_EXTERNAL)
 target_compile_options(${target} PRIVATE -include "${ROOT}/src/Share/fmt_enum_compat.h")
 target_link_directories(${target} PRIVATE /opt/homebrew/lib)
 target_link_libraries(${target} PRIVATE boost_filesystem boost_thread pthread dl /usr/local/lib/libiconv.dylib)
endforeach()
target_link_options(i03_parser_shm_v2 PRIVATE -fprofile-instr-generate)
target_link_libraries(i03_parser_shm_v2 PRIVATE "${WT_BUILD}/build_x64/Debug/libs/libWtCore.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSTools.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSUtils.a" /opt/homebrew/lib/libgtest_main.a /opt/homebrew/lib/libgtest.a)
target_link_options(i03_parser_shm_legacy PRIVATE -fprofile-instr-generate)
target_link_libraries(i03_parser_shm_legacy PRIVATE "${WT_BUILD}/build_x64/Debug/libs/libWtCore.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSTools.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSUtils.a" /opt/homebrew/lib/libgtest_main.a /opt/homebrew/lib/libgtest.a)
