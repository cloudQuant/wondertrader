function(wt_fixture name sanitizer)
  add_executable(${name} ${ARGN})
  target_include_directories(${name} BEFORE PRIVATE ${INCS})
  target_compile_definitions(${name} PRIVATE FMT_HEADER_ONLY SPDLOG_FMT_EXTERNAL)
  target_compile_options(${name} PRIVATE "SHELL:-include cstdlib" "SHELL:-include ${ROOT}/src/Share/fmt_enum_compat.h"
    -g -O1 -fno-omit-frame-pointer -fsanitize=${sanitizer})
  target_link_options(${name} PRIVATE -fsanitize=${sanitizer} -fprofile-instr-generate)
  target_link_directories(${name} PRIVATE /opt/homebrew/lib /usr/local/lib)
  if(sanitizer STREQUAL "address,undefined")
    target_compile_options(${name} PRIVATE -fno-sanitize-recover=undefined)
  endif()
endfunction()
