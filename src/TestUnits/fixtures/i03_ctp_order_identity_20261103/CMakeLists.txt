set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(WT_DEP_BUILD "" CACHE PATH "Independently built WonderTrader directory")

target_include_directories(i03_ctp_identity PRIVATE /opt/homebrew/Cellar/boost/1.88.0/include /opt/homebrew/include /opt/local/include)
