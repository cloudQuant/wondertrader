cmake_minimum_required(VERSION 3.16)
project(i03_servo_date LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(WT_BUILD "" CACHE PATH "Independent build containing WTSTools and WTSUtils")
set(INCS "${ROOT}/src" /opt/homebrew/Cellar/boost/1.88.0/include /opt/homebrew/include /usr/local/include)
add_library(DateSpy SHARED reader.cpp)
target_include_directories(DateSpy PRIVATE ${INCS})
target_compile_definitions(DateSpy PRIVATE FMT_HEADER_ONLY SPDLOG_FMT_EXTERNAL)
target_compile_options(DateSpy PRIVATE "SHELL:-include cstdlib" "SHELL:-include ${ROOT}/src/Share/fmt_enum_compat.h" -fsanitize=undefined -fno-sanitize-recover=all)
target_link_options(DateSpy PRIVATE -fsanitize=undefined)
add_executable(servo_date check.cpp "${ROOT}/src/WtDtServo/WtDataManager.cpp" "${ROOT}/src/WtDtServo/WtDtRunner.cpp" "${ROOT}/src/WtDtServo/ParserAdapter.cpp" "${ROOT}/src/WtDtServo/WtHelper.cpp" "${ROOT}/src/WTSTools/WTSBaseDataMgr.cpp")
target_include_directories(servo_date PRIVATE ${INCS})
target_compile_definitions(servo_date PRIVATE FMT_HEADER_ONLY SPDLOG_FMT_EXTERNAL)
target_compile_options(servo_date PRIVATE "SHELL:-include cstdlib" "SHELL:-include ${ROOT}/src/Share/fmt_enum_compat.h" -fsanitize=undefined -fno-sanitize-recover=all)
target_link_options(servo_date PRIVATE -fsanitize=undefined -fprofile-instr-generate)
target_link_directories(servo_date PRIVATE /opt/homebrew/lib /usr/local/lib)
target_link_libraries(servo_date PRIVATE "${WT_BUILD}/build_x64/Debug/libs/libWTSTools.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSUtils.a" /opt/homebrew/lib/libgtest_main.a /opt/homebrew/lib/libgtest.a boost_filesystem boost_thread pthread dl /usr/local/lib/libiconv.dylib)
