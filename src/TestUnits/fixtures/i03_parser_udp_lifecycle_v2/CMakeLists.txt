cmake_minimum_required(VERSION 3.16)
project(i03_parser_udp_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(WT_BUILD "" CACHE PATH "Build providing static dependencies")
add_library(ParserUDP SHARED "${ROOT}/src/ParserUDP/ParserUDP.cpp")
set_target_properties(ParserUDP PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/parsers")
add_library(ParserUDPFault SHARED "${ROOT}/src/ParserUDP/ParserUDP.cpp")
target_compile_definitions(ParserUDPFault PRIVATE WT_PARSERUDP_TEST_HOOKS)
set_target_properties(ParserUDPFault PROPERTIES OUTPUT_NAME ParserUDP LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fault/parsers")
add_executable(i03_parser_udp_v2 check.cpp "${ROOT}/src/WtCore/ParserAdapter.cpp")
add_executable(i03_parser_udp_legacy "${ROOT}/src/ParserUDP/ParserUDP.cpp" "${ROOT}/src/WtCore/ParserAdapter.cpp" "${ROOT}/src/TestUnits/coverage/t_ParserUDP/r76_udp_boundary_test.cpp" "${ROOT}/src/TestUnits/coverage/t_ParserUDP/t6_parserudp_reachable_test.cpp" "${ROOT}/src/TestUnits/coverage/t_ParserUDP/udp_handle_write_lock_test.cpp")
foreach(target ParserUDP ParserUDPFault i03_parser_udp_v2 i03_parser_udp_legacy)
 target_include_directories(${target} PRIVATE "${ROOT}/src" /opt/homebrew/include /opt/homebrew/Cellar/boost/1.88.0/include)
 target_compile_definitions(${target} PRIVATE FMT_HEADER_ONLY SPDLOG_FMT_EXTERNAL)
 target_compile_options(${target} PRIVATE -include "${ROOT}/src/Share/fmt_enum_compat.h")
 target_link_directories(${target} PRIVATE /opt/homebrew/lib)
 target_link_libraries(${target} PRIVATE boost_filesystem boost_thread pthread dl /usr/local/lib/libiconv.dylib)
endforeach()
target_link_options(i03_parser_udp_v2 PRIVATE -fprofile-instr-generate)
target_link_libraries(i03_parser_udp_v2 PRIVATE "${WT_BUILD}/build_x64/Debug/libs/libWtCore.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSTools.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSUtils.a" /opt/homebrew/lib/libgtest_main.a /opt/homebrew/lib/libgtest.a)
target_link_options(i03_parser_udp_legacy PRIVATE -fprofile-instr-generate)
target_link_libraries(i03_parser_udp_legacy PRIVATE "${WT_BUILD}/build_x64/Debug/libs/libWtCore.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSTools.a" "${WT_BUILD}/build_x64/Debug/libs/libWTSUtils.a" /opt/homebrew/lib/libgtest_main.a /opt/homebrew/lib/libgtest.a)
